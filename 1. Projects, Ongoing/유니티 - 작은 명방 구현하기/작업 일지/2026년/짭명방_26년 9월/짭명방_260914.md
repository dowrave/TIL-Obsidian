### 화면 반댓쪽의 엣지들도 보이게 하기

#### 원인
![[Pasted image 20260914151219.png]]
- `Saturate` 노드는 0 미만의 값을 0으로 보정하고, 1을 초과하는 값을 1로 보정한다.
- `Dot Product, 내적` 노드의 출력 결과값은 `-1, 1`인데 0 미만을 잘라내서 저렇게 보이는 것.

#### 수정
![[Pasted image 20260914153042.png]]
- `Saturate`을 제거 : `[-1, 0]`도 정보로 활용
	- 이에 따라 `Remap`의 인풋을 `[-1, 1]`로 수정
- `Lerp` 노드 추가 : `BackColor, FrontColor`를 프로퍼티로 추가.
	- 별도로 계산됐던 `VertexColor`에 계산되는 `Color` 노드를 없애고, 위의 `FrontColor`를 해당 개념으로 사용함
	- `FrontColor` : 흰색(혹은 사용하고 싶은 색)
	- `BackColor` : 검은색, 알파 0
		- 뒷면이 서서히 보이지 않게 됨

### 일부 버텍스 어둡게 처리하는 거 적용 안됨
- 대충 이런 코드를 쓰고 있었음. 
```cs
// 구 표면을 연속적인 Perlin Noise로 흑백 무늬처럼 칠한 다음, 일정 밝기 이하인 부분만 어둡게 깎아내 패치로 만든다.
public static Color[] BuildSelectiveDarkeningColors(List<Vector3> verts, float noiseScale, float darkThreshold, float darkAmount, int seed = 0)
{
	var colors = new Color[verts.Count];

	// 시드값을 노이즈 샘플링 좌표에 오프셋으로 넣어서 시드가 다르면 패치 위치가 달라지게 함
	Vector3 offset = new Vector3(seed * 17.13f, seed * 5.71f, seed * 31.9f);

	for (int i = 0; i < verts.Count; i++)
	{
		// 1. 정점 위치를 노이즈 좌표로 변환
		Vector3 p = verts[i].normalized * noiseScale + offset;

		// 2. 2D Perlin Noise를 2번 조합, 3D 노이즈를 흉내냄
		float n1 = Mathf.PerlinNoise(p.x, p.y); // (x, y) 평면 샘플링
		float n2 = Mathf.PerlinNoise(p.y, p.z); // (y, z) 평면 샘플링
		float noise = (n1 + n2) * 0.5f; // 두 값을 평균 냄

		// 3. 노이즈 값이 임계치보다 낮은 영역을 "어두운 패치"로 판정
		float brightness = noise < darkThreshold ? darkAmount : 1f;

		// 4. RGB 동일 값으로 흑백 밝기 색 생성. 셰이더에 VertexColor로 들어가는 값.
		colors[i] = new Color(brightness, brightness, brightness, 1f);
	}
	return colors;
}
```

### 엣지 간 통신 시각적 효과
- 도형의 각 엣지에 통신이 되고 있다는 효과를 주고 싶음
- 근데 신호가 흐른다 = 방향이 있다 = 엣지의 각 부분(프래그먼트)마다 0 ~ 1 사이의 값이 있다 인데, 현재의 구현은 정점 하나가 여러 엣지에 동시에 참여하는 구조임
- 따라서, 각 정점의 위치를 기반으로 엣지를 만들되, 그 엣지들은 각각의 고유한 정점을 갖는 식으로 구현됨.
	- **원본 정점**이 있고, **엣지 전용 정점**을 같은 위치에 새로운 정점을 넣는 개념임.
	- 그래서 **엣지 갯수 * 2 만큼 버텍스 갯수가 추가**된다.

#### 구현 - 코드
```cs
// 엣지마다 버텍스 2개를 만들어, UV.x에 엣지 내 위치(t: 0 ~ 1), UV.y에 엣지 고유 랜덤 시드를 저장하는 라인 메쉬
// 셰이더에서 엣지 별로 독립된 흐름 효과를 계산할 수 있게끔 한다.
public static Mesh BuildFlowLineMesh(
	List<Vector3> verts,
	List<(int, int)> edges,
	Color[] vertexColors,
	Vector3[] normals,
	int seed = 0)
{
	// 엣지마다 고유한 2개의 정점을 가져야 하므로, 관련 요소들도 다 2배로 구현
	var newVerts = new List<Vector3>(edges.Count * 2);
	var newColors = new List<Color>(edges.Count * 2);
	var newNormals = new List<Vector3>(edges.Count * 2);
	var newUVs = new List<Vector2>(edges.Count * 2);
	var indices = new List<int>(edges.Count * 2);

	// 랜덤 생성기. 
	// UnityEngine.Random은 전역 상태를 공유하므로 해당 함수가 몇 번 호출됐느냐에 따라 값이 달라질 수 있다.
	// System.Random은 함수 안에서만 독립적으로 동작하므로 재현 가능한 패턴을 만들 수 있다.
	var rng = new System.Random(seed);

	foreach (var (a, b) in edges)
	{
		// 엣지별 시드. 양쪽 정점이 동일해야 보간돼도 엣지 전체에서 같은 값을 유지함
		float edgeRandom = (float)rng.NextDouble();

		// 설정할 때마다 newVerts에 값을 추가하므로, 이는 고유의 인덱스가 됨
		int baseIndex = newVerts.Count;

		// 원본 정점의 위치, 색상, 노멀을 복사해 새 정점을 만든 후 UV(0, random)을 붙임.
		// b도 마찬가지이며, UV(1, random)이 들어간다.
		newVerts.Add(verts[a]);
		newColors.Add(vertexColors != null ? vertexColors[a] : Color.white);
		newNormals.Add(normals != null ? normals[a] : Vector3.zero);
		newUVs.Add(new Vector2(0f, edgeRandom)); // t = 0 (시작점)

		newVerts.Add(verts[b]);
		newColors.Add(vertexColors != null ? vertexColors[b] : Color.white);
		newNormals.Add(normals != null ? normals[b] : Vector3.zero);
		newUVs.Add(new Vector2(1f, edgeRandom)); // t = 1 (도착점)

		// 인덱스 버퍼에 a, b 두 정점의 인덱스를 추가.
		indices.Add(baseIndex);
		indices.Add(baseIndex + 1);
	}

	// 메쉬 조립
	var mesh = new Mesh(); // 기본적으로 정점 인덱스를 16비트로 저장함
	if (newVerts.Count > 65535)
		mesh.indexFormat = UnityEngine.Rendering.IndexFormat.UInt32; // 정점이 2배로 늘어나므로 한도 체크

	// 리스트를 배열로 저장해서 메쉬에 채워넣음.
	mesh.vertices = newVerts.ToArray();
	mesh.colors = newColors.ToArray();
	mesh.normals = newNormals.ToArray();
	mesh.uv = newUVs.ToArray();
	mesh.SetIndices(indices.ToArray(), MeshTopology.Lines, 0);

	return mesh;
}
```
