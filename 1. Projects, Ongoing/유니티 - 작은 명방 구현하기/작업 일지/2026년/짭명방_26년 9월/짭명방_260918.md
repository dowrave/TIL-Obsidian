## TitleScene 아트 작업 신호 크기 조절

### 기존 구현과 해결 방향
- 기존 방식
	- `MeshTopology.Lines`로 메쉬를 만들고 있음.
	- `Lines` 토폴로지는 GPU가 선분을 그릴 때 쓰는 별도의 렌더링 모드이며, 이 경우 셰이더 그래프의 `Vertex / Fragment` 단계에서는 선의 폭에 관여할 방법이 없다.
	- 즉, **지오메트리 자체가 폭이 없는 선**인 상황. 

- 해결 방향
	- **엣지를 실제 삼각형 리본으로 만드는 쪽으로 구조를 바꿔야 한다.**

### 구현 원리

#### 기존 : MeshTopology.Lines
- `MeshTopology.Lines`로 그리는 선은 면(삼각형)이 아니라 1차원 선분이다. 
- **GPU의 입장에서 삼각형은 내부를 채워넣는 래스터라이징 과정이 있지만, 선분은 면적이라는 개념이 없다.**
- 따라서 **화면에 선분을 그릴 때 GPU/드라이버가 선분 주변의 몇 픽셀까지를 선으로 칠할지를 자체적으로 정한다.** 이 과정은 셰이더가 아니라 GPU/드라이버에서 결정된다.
- 그렇기 때문에, 버텍스 셰이더는 점이 어디 있는가만 나타낼 수 있고, 점 주변에 얼마나 넓게 칠해야 하는가에 관한 정보는 알 수 없다. 폭이라는 정보가 지오메트리 안에 없기 때문에, 셰이더 그래프에서 조절할 수 있는 개념이 아니게 된다.

#### 변경 : 리본 메쉬
- 리본은 선이 아니라 **폭을 가진 얇고 긴 평면(면적이 있는 사각형)** 이다.  
- 겉보기엔 얇은 선으로 보이나, **원래 선분의 양 옆으로 살짝 벌어진 2개의 점을 만들어서 그 4개의 점을 삼각형 2개로 채운 것**이다.
- 따라서 점들이 벌어진 정도에 관한 위치 계산은 버텍스 셰이더가 다루는 값이 된다. 

![[Pasted image 20260918161219.png]]
- 리본 방식은 원래의 정점 위치에서 좌우로 벌어진 4개의 점을 만들고, 그 4개의 점으로 삼각형 2개를 만든다. (우상 - 좌하로 자름)
- 여기서 벌어지는 정도를 셰이더의 `EdgeThickness`로 조절할 수 있다는 의미. 

> 중간에 CSS 만지느라 시간이 좀 걸렸다. 

### 구현
- `TitleSolid.BuildFlowLineMesh`
```cs
// 엣지마다 리본(폭이 있는 사각형 = 2개의 삼각형)을 만드는 메쉬.
// Tangent.xyz = 엣지에 수직인 방향(perp(endicular)). 셰이더 버텍스에서 Position += Tangent.xyz * UV.z * (EdgeThickness * 0.5) 형태로 오프셋해서 실제 두께를 만듦
public static Mesh BuildFlowLineMesh(
	List<Vector3> verts,
	List<(int, int)> edges,
	Color[] vertexColors,
	Vector3[] normals,
	int seed = 0)
{
	// 엣지마다 고유한 2개의 정점을 가져야 하므로, 관련 요소들도 다 2배로 구현
	// var newVerts = new List<Vector3>(edges.Count * 2);
	// var newColors = new List<Color>(edges.Count * 2);
	// var newNormals = new List<Vector3>(edges.Count * 2);
	// var newUVs = new List<Vector2>(edges.Count * 2);
	// var indices = new List<int>(edges.Count * 2);

	// 엣지마다 고유한 4개의 정점(a+, a-, b+, b-)을 가져야 하므로, 관련 요소들도 다 4배로 구현
	var newVerts = new List<Vector3>(edges.Count * 4);
	var newColors = new List<Color>(edges.Count * 4);
	var newNormals = new List<Vector3>(edges.Count * 4);
	
	var newUVs = new List<Vector3>(edges.Count * 4); // UV.x : 엣지의 위치(t = 0 ~ 1) / UV.y = 엣지 고유 랜덤 시드, UV.z = 리본 내 side(+1 / -1)
	var newTangents = new List<Vector4>(edges.Count * 4); // 리본으로 구현하면서 추가된 요소
	var indices = new List<int>(edges.Count * 6); // 2개의 삼각형을 그려야 하므로 메쉬당 6개의 인덱스가 필요(겹치더라도)

	// 랜덤 생성기. 
	// UnityEngine.Random은 전역 상태를 공유하므로 해당 함수가 몇 번 호출됐느냐에 따라 값이 달라질 수 있다.
	// System.Random은 함수 안에서만 독립적으로 동작하므로 재현 가능한 패턴을 만들 수 있다.
	var rng = new System.Random(seed);

	foreach (var (a, b) in edges)
	{
		// 엣지별 시드. 양쪽 정점이 동일해야 보간돼도 엣지 전체에서 같은 값을 유지함
		float edgeRandom = (float)rng.NextDouble();

		// 엣지 방향 및 양 끝 정점 노멀을 평균한 대표 노멀, 엣지에 수직인 방향(perp)을 구함
		Vector3 edgeDir = (verts[b] - verts[a]).normalized;
		Vector3 midNormal = (normals[a] + normals[b]).normalized;
		Vector3 perp = Vector3.Cross(edgeDir, midNormal);

		// edgeDir과 midNormal이 거의 평행한 경우 Cross가 0에 가까워지는 예외 상황 방지
		if (perp.sqrMagnitude < 1e-6f)
		{
			perp = Vector3.Cross(edgeDir, Vector3.up);
			if (perp.sqrMagnitude < 1e-6f) perp = Vector3.right;
		}

		perp.Normalize();
		Vector4 perp4 = new Vector4(perp.x, perp.y, perp.z, 0f);

		Color colorA = vertexColors != null ? vertexColors[a] : Color.white;
		Color colorB = vertexColors != null ? vertexColors[b] : Color.white;
		Vector3 normalA = normals != null ? normals[a] : Vector3.zero; 
		Vector3 normalB = normals != null ? normals[b] : Vector3.zero;

		// 설정할 때마다 newVerts에 값을 추가하므로, 이는 고유의 인덱스가 됨
		int baseIndex = newVerts.Count;

		// newVerts.Add(verts[a]);
		// newColors.Add(vertexColors != null ? vertexColors[a] : Color.white);
		// newNormals.Add(normals != null ? normals[a] : Vector3.zero);
		// newUVs.Add(new Vector2(0f, edgeRandom));

		// newVerts.Add(verts[b]);
		// newColors.Add(vertexColors != null ? vertexColors[b] : Color.white);
		// newNormals.Add(normals != null ? normals[b] : Vector3.zero);
		// newUVs.Add(new Vector2(1f, edgeRandom));

		// indices.Add(baseIndex);
		// indices.Add(baseIndex + 1);


		// a_+, a_-, b_+, b_- 순서로 4개의 정점을 생성함
		newVerts.Add(verts[a]); 
		newColors.Add(colorA); 
		newNormals.Add(normalA);
		newUVs.Add(new Vector3(0f, edgeRandom, +1f));
		newTangents.Add(perp4);

		newVerts.Add(verts[a]); 
		newColors.Add(colorA); 
		newNormals.Add(normalA);
		newUVs.Add(new Vector3(0f, edgeRandom, -1f));
		newTangents.Add(perp4);

		newVerts.Add(verts[b]); 
		newColors.Add(colorB); 
		newNormals.Add(normalB);
		newUVs.Add(new Vector3(1f, edgeRandom, +1f));
		newTangents.Add(perp4);

		newVerts.Add(verts[b]); 
		newColors.Add(colorB); 
		newNormals.Add(normalB);
		newUVs.Add(new Vector3(1f, edgeRandom, -1f));
		newTangents.Add(perp4);

		// indices.Add(baseIndex);
		// indices.Add(baseIndex + 1);

		// 쿼드를 삼각형 2개로 분할
		indices.Add(baseIndex + 0);
		indices.Add(baseIndex + 2);
		indices.Add(baseIndex + 1);

		indices.Add(baseIndex + 1);
		indices.Add(baseIndex + 2);
		indices.Add(baseIndex + 3);
	}

	// 메쉬 조립
	var mesh = new Mesh(); // 기본적으로 정점 인덱스를 16비트로 저장함
	if (newVerts.Count > 65535)
		mesh.indexFormat = UnityEngine.Rendering.IndexFormat.UInt32; // 정점이 2배로 늘어나므로 한도 체크

	// 리스트를 배열로 저장해서 메쉬에 채워넣음.
	mesh.vertices = newVerts.ToArray();
	mesh.colors = newColors.ToArray();
	mesh.normals = newNormals.ToArray();
	
	// mesh.uv = newUVs.ToArray();
	// mesh.SetIndices(indices.ToArray(), MeshTopology.Lines, 0);
	mesh.SetUVs(0, newUVs);
	mesh.tangents = newTangents.ToArray();
	mesh.SetIndices(indices.ToArray(), MeshTopology.Triangles, 0);

	return mesh;
}
```
> 주석 처리된 부분은 기존의 Line 기반, 수정된 부분은 그 아래에 작성됨.

- 삼각형을 만드는 부분
	- 대각선을 어떤 방향으로 잡아주느냐에 따라 크게 2가지 방법이 있는데, 어느 쪽을 골라도 사각형 전체를 빈틈없이 채울 수 있다.
	- 유니티는 **카메라가 보는 방향에서 정점을 나열한 순서가 시계 방향이라면 앞면(=보이는 면)이라는 규칙을 쓴다.** (OpenGL 계열은 반시계 = 앞면이라는 차이점이 있음)
		- 카메라가 보는 방향 = 이 케이스에서는 구 바깥쪽에서 표면을 들여다보는 방향.
		- `a+, b+, a-` : 좌상 - 좌하 - 우상, 반시계 방향
		- `a-, b+, b-` : 우상 - 좌하 - 우하, 반시계 방향
