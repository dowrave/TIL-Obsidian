## 데이터 저장 PlayerPrefs에서 변경

- 어제 공부했던 [[유니티 - 세이브 시스템 알아보기|유니티 세이브 시스템]]에 대해, 무지성으로 사용했던 **`PlayerPrefs`가 좋은 요소가 아니라고 배웠음.**
	- `PlayerPrefs`는 `KV(Key-Value)` 저장소라서(Windows - 레지스트리, macOS - plist 등) 구조화된 세이브 데이터를 넣으면 문제가 발생할 수 있음
		- 문자열 크기, 개수에 대한 암묵적 제한 / 플랫폼별 저장 방식 차이
		- 세이브 파일을 따로 백업/복사/디버깅하기 어려움
		- 이후 슬롯별 파일 관리, 클라우드 동기화 등을 붙이기 번거로움
	- 물론 이런 규모의 프로젝트에선 쓸 수도 있다고 생각하지만, **이왕 할 거 제대로 된 방식도 써보자**는 생각.

- 일반적으로 세이브파일은 `Application.persistentDataPath`에 `System.IO.File`로 읽고 쓴다. 
	- `Application.persistentDataPath` : 플랫폼별로 안전하게 쓰기 위한 디렉토리.

### 1. 저장 키 -> 저장 경로로 변경
```cs
// 변경 전
private string SaveKey => $"PlayerData_Slot{_slotId}";

// 변경 후
private string SavePath => Path.Combine(Application.persistentDataPath, $"PlayerData_Slot{_slotId}.json");
```

- 기존의 `SetString()`은, 여러 필드와 리스트가 얽힌 하나의 큰 구조체를 통째로 JSON 문자열 하나로 뭉쳐서, `PlayerData_Slot{index}`의 형태로 전달하고 있었다.
- KV 저장소에서는 이 JSON 문자열이 그 안에 얼마나 들었고 얼마나 큰지 잘 모름.
- 반면 파일 시스템의 `.json` 파일로 저장하면 파일 하나 = 세이브 데이터 하나로 그 형태가 자연스럽게 맞아떨어진다.

## 다른 코드에서 PlayerPrefs을 사용하던 부분들도 수정
- 옵션 설정은 여기에 해당되지 않는다 - 기존의 `PlayerPrefs`을 사용하는 형태를 유지해도상관 없음.

### `MainMenuManager` 
- 3성 클리어하지 못한, 스테이지의 Id를 종료 시점에 저장하고 메인 메뉴로 돌아왔을 때 해당 스테이지 버튼이 선택되도록 하는 로직이 있었다.  여기서 `PlayerPrefs`을 썼음. 
- 이 매니저에서 `PlayerPrefs`로 직접 데이터를 불러오는 과정 또한 이상했기 때문에 관련 필드와 로직을 `PlayerSessionDataManager`로 옮겼음.
- 여기서의 필드는 세이브 데이터에 저장될 필요는 없는 개념. `PlayerData`에 넣진 않았다. 

### SaveSlotMetadataStore
```cs
using System.IO;
using System;
using UnityEngine;

// 슬롯별 SaveSlotMetadata를 저장/조회하는 유틸리티.
// PlayerSessionDataManager(쓰기), SaveSlotManager(읽기) 양쪽에서 공통으로 사용한다.
public static class SaveSlotMetadataStore
{
    // private const string MetaKeyPrefix = "SaveSlotMeta_Slot";
    private const string MetaFilePrefix = "SaveSlotMeta_Slot";
    
    // private static string KeyFor(int slotId) => $"{MetaKeyPrefix}{slotId}";
    private static string PathFor(int slotId) => Path.Combine(Application.persistentDataPath, $"{MetaFilePrefix}{slotId}.json");
    // public static bool Exists(int slotId) => PlayerPrefs.HasKey(PathFor(slotId));
    public static bool Exists(int slotId) => File.Exists(PathFor(slotId));
    
    public static SaveSlotMetadata Load(int slotId)
    {
        // string json = PlayerPrefs.GetString(KeyFor(slotId), "");
        // if (string.IsNullOrEmpty(json)) return null;

        // return JsonUtility.FromJson<SaveSlotMetadata>(json);

        string path = PathFor(slotId);
        if (!File.Exists(path)) return null;

        try
        {
            string json = File.ReadAllText(path);
            return JsonUtility.FromJson<SaveSlotMetadata>(json);
        }
        catch (Exception e)
        {
            Logger.LogError($"슬롯 {slotId} 메타데이터 로드 실패 : {e.Message}");
            return null;
        }
    
    }

    public static void Save(int slotId, SaveSlotMetadata metadata)
    {
        string json = JsonUtility.ToJson(metadata);
        // PlayerPrefs.SetString(KeyFor(slotId), json);
        try
        {
            File.WriteAllText(PathFor(slotId), json);
        }
        catch (Exception e)
        {
            Logger.LogError($"슬롯 {slotId} 메타데이터 저장 실패 : {e.Message}");
        }
    }

    public static void Delete(int slotId)
    {
        // PlayerPrefs.DeleteKey(KeyFor(slotId));
        string path = PathFor(slotId);
        if (File.Exists(path))
        {
            File.Delete(path);
        }
    }
}

```
### SaveSlotManager
```cs
using System.Collections.Generic;
using System.IO;
using UnityEngine;

// 타이틀 화면에서 사용하는 매니저. 
// 3개 슬롯의 요약 정보를 보여주고 슬롯을 선택해 게임 씬으로 진입시키는 역할.
// 실제 데이터의 로드 / 생성은 PlayerSessionDataManager 담당.
public class SaveSlotManager : MonoBehaviour
{
    public const int SlotCount = 3;
    // private const string DataKeyPrefix = "PlayerData_Slot";
    private const string DataFilePrefix = "PlayerData_Slot";

    public static SaveSlotManager Instance { get ; private set; }

    public int SelectedSlotId { get; private set; } = 0;

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(this);
            return;
        }

        Instance = this;
        // DontDestroyOnLoad(gameObject);  // 필요한가?
    }

    private string DataPathFor(int slotId)
        => Path.Combine(Application.persistentDataPath, $"{DataFilePrefix}{slotId}.json");

    // 해당 슬롯의 세이브 데이터 여부 확인
    // public bool DoesSlotExist(int slotId) => PlayerPrefs.HasKey(DataKeyPrefix + slotId);
    public bool DoesSlotExist(int slotId) => File.Exists(DataPathFor(slotId));
    
    // 슬롯 하나의 요약 정보
    public SaveSlotMetadata GetSlotMetadata(int slotId) => SaveSlotMetadataStore.Load(slotId);

    // 슬롯 선택 UI에 뿌릴 3칸짜리 리스트
    public List<SaveSlotMetadata> GetAllSlotMetadata()
    {
        var result = new List<SaveSlotMetadata>(SlotCount);

        for (int i = 0; i < SlotCount; i++)
        {
            result.Add(GetSlotMetadata(i));
        }
        return result; 
    }

    // 슬롯 삭제
    public void DeleteSlot(int slotId)
    {
        // PlayerPrefs.DeleteKey(DataKeyPrefix + slotId);
        // SaveSlotMetadataStore.Delete(slotId);
        // PlayerPrefs.Save();

        string path = DataPathFor(slotId);
        if (File.Exists(path))
        {
            File.Delete(path);
        }
        SaveSlotMetadataStore.Delete(slotId);
    }

    // 슬롯 선택 후 게임 씬 진입
    public void SelectSlotAndEnterGame(int slotId)
    {
        SelectedSlotId = slotId;
        SceneLoader sceneLoader = GameManagement.Instance.SceneLoader;
        if (sceneLoader != null)
        {
            sceneLoader.LoadMainMenu();
        }
    }
}

```

### `SaveSlot`이 저장되는 경로에 관한 문제 
- 현재는 양쪽(세션 데이터 매니저, 세이브 슬롯 매니저)에서 관리되고 있음.
- 이를 **세이브 슬롯 매니저로 통일**함(게임 내내 띄워져 있기 때문. 세션 데이터 매니저는 세션 단위로만 띄워져 있다.)

```cs
public static string GetPlayerDataPath(int slotId)
	=> Path.Combine(Application.persistentDataPath, $"{DataFilePrefix}{slotId}.json"); // 경로 규칙은 여기서 관리
```

- 다른 스크립트 `PlayerSessionDataManager`에선 아래처럼 접근함.
```cs
// private string SavePath => Path.Combine(Application.persistentDataPath, $"PlayerData_Slot{_slotId}.json");
private string SavePath => SaveSlotManager.GetPlayerDataPath(_slotId); 
```

> `SaveSlotManager`는 싱글턴 인스턴스지만, `GetPlayerDataPath`는 순수 계산 함수에 가깝기 때문에 `static`으로 구현되었다. 단순 메서드로 구현해도 크게 상관 없다. 초기화 순서를 신경써야 한다는 점이 이슈가 될 수 있다는 정도?
