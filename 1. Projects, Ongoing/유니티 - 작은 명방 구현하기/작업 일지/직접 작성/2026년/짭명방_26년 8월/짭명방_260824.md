2. `SaveSlotMetadataStore.cs`
```cs
using UnityEngine;

// 슬롯별 SaveSlotMetadata를 PlayerPrefs에 저장/조회하는 유틸리티.
// PlayerSessionDataManager(쓰기), SaveSlotManager(읽기) 양쪽에서 공통으로 사용한다.
public static class SaveSlotMetadataStore
{
    private const string MetaKeyPrefix = "SaveSlotMeta_Slot";
    
    private static string KeyFor(int slotId) => $"{MetaKeyPrefix}{slotId}";
    public static bool Exists(int slotId) => PlayerPrefs.HasKey(KeyFor(slotId));

    public static SaveSlotMetadata Load(int slotId)
    {
        string json = PlayerPrefs.GetString(KeyFor(slotId), "");
        if (string.IsNullOrEmpty(json)) return null;

        return JsonUtility.FromJson<SaveSlotMetadata>(json);
    }

    public static void Save(int slotId, SaveSlotMetadata metadata)
    {
        string json = JsonUtility.ToJson(metadata);
        PlayerPrefs.SetString(KeyFor(slotId), json);
    }

    public static void Delete(int slotId)
    {
        PlayerPrefs.DeleteKey(KeyFor(slotId));
    }
}
```

3. `SaveSlotManager.cs`
```cs
using System.Collections.Generic;
using UnityEngine;

// 타이틀 화면에서 사용하는 매니저. 
// 3개 슬롯의 요약 정보를 보여주고 슬롯을 선택해 게임 씬으로 진입시키는 역할.
// 실제 데이터의 로드 / 생성은 PlayerSessionDataManager 담당.
public class SaveSlotManager : MonoBehaviour
{
    public const int SlotCount = 3;
    private const string DataKeyPrefix = "PlayerData_Slot";

    public static SaveSlotManager Instance { get ; private set; }

    public int SelectedSlotId { get; private set; } = 0;

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(this);
            return;
        }

        Instance = this;
        // DontDestroyOnLoad(gameObject);  // 필요한가?
    }

    // 해당 슬롯의 세이브 데이터 여부 확인
    public bool DoesSlotExist(int slotId) => PlayerPrefs.HasKey(DataKeyPrefix + slotId);

    // 슬롯 하나의 요약 정보
    public SaveSlotMetadata GetSlotMetadata(int slotId) => SaveSlotMetadataStore.Load(slotId);

    // 슬롯 선택 UI에 뿌릴 3칸짜리 리스트
    public List<SaveSlotMetadata> GetAllSlotMetadata()
    {
        var result = new List<SaveSlotMetadata>(SlotCount);

        for (int i = 0; i < SlotCount; i++)
        {
            result.Add(GetSlotMetadata(i));
        }
        return result; 
    }

    // 슬롯 삭제
    public void DeleteSlot(int slotId)
    {
        PlayerPrefs.DeleteKey(DataKeyPrefix + slotId);
        SaveSlotMetadataStore.Delete(slotId);
        PlayerPrefs.Save();
    }

    // 슬롯 선택 후 게임 씬 진입
    public void SelectSlotAndEnterGame(int slotId)
    {
        SelectedSlotId = slotId;
        SceneLoader sceneLoader = GameManagement.Instance.SceneLoader;
        if (sceneLoader != null)
        {
            sceneLoader.LoadMainMenu();
        }
    }
}
```

#### 누적 시간 계산
- [[유니티 - 플레이타임 누적 시간 저장]]
- **`Update()`에서 `Time.deltaTime`을 누적하지 않는다.** 
- 세션이 시작된 시각만 저장한 다음, 저장 시점마다 흐른 시간을 한꺼번에 더하는 방식이다.

1. 세션 시작 시점 - 새로 시작하기, 불러오기 시점
```cs
// PlayerSessionDataManager.cs
private void BeginPlaytimeTracking()
{
	_sessionStartRealtime = Time.realtimeSinceStartup;
}
```

2. 커밋 시점 - `SavePlayerData()` 내부에 있는 코드
```cs
private void CommitPlaytime()
{
	if (playerData == null) return;
	float elapsed = Time.realtimeSinceStartup - _sessionStartRealtime;
	if (elapsed > 0f)
	{
		playerData.totalPlayTimeSeconds += elapsed; 
	}
	
	// 저장 후 다시 현재 시점을 세션 시작 시간으로 설정(이전 시간은 누적 시간에 반영됐으니까)
	_sessionStartRealtime = Time.realtimeSinceStartup;
}
```

3. 실시간 조회
```cs
public float GetTotalPlayTimeSeconds()
{
	if (playerData == null) return 0f;
	return playerData.totalPlayTimeSeconds + (Time.realtimeSinceStartup - _sessionStartRealtime);
}
```

- 이런 식으로 구현된 이유 : `SavePlayerData()`의 빈도가 잦기 때문이다.

1. `OnApplicationPause(bool pauseStatus)`
	- **앱이 일시정지 상태에 들어가거나 나올 때 호출**되는 유니티 생명주기 콜백.
	- 모바일에서 중요하다. 백그라운드로 진입하면 `true`, 다시 창을 활성화하면 `false`가 호출된다.
	- PC에서는 크게 중요하진 않은데, 그렇다고 해서 절대 호출되지 않는다는 개념은 아니다. 
		- 유니티 에디터 / 플레이 모드에서의 앱 일시정지
		- 일부 플랫폼에서 앱이 백그라운드 상태로 전환
		- 운영체제나 실행 환경이 앱을 suspend함

2. `OnApplicationQuit()`
	- **앱이 정상적으로 종료될 때 유니티가 호출**하는 콜백. 
	- `Alt + F4` 등으로 게임이 꺼질 때도 동작한다.
	- 하지만 프로세스 강제 종료, PC 전원 강제 종료, 크래시 등의 상황에서는 마지막 저장이 보장되지 않아서, `Save 버튼 + 자동 저장 + OnApplicationQuit()` 등의 다중 안전장치를 쓰는 게 좋다. 

- 이외에도 `OnApplicationFocus(bool hasFocus)`도 있다.
	- 알탭 등으로 포커스를 얻거나 잃었을 때 동작하는 로직. 

여기서는 `OnApplicationPause`나 `OnApplicationFocus` 양쪽 모두 크게 필요해보이는 동작은 아니라서 주석처리했다. `OnApplicationQuit()`만 사용하는 것으로.
