- 현재 하이어라키에서는 아래의 구조를 띄고 있으며, `SolidRoot`가 사실상 하는 게 없다.

#### SolidBuilder.cs
```cs
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

[RequireComponent(typeof(MeshFilter), typeof(MeshRenderer))]
public class SolidBuilder : MonoBehaviour
{
    [Header("Solid 설정")]
    [SerializeField] private float solidScale = 2f;
    [SerializeField] private Material lineMaterial;
    [SerializeField] private int subdivisions = 1;

    [Header("Solid - VertexPulse 설정")]
    [SerializeField] private Material pulseMaterial;
    [SerializeField] private int circleSegments = 24;
    [SerializeField] private Camera targetCamera;
    [SerializeField] private float pulseMaxRadius = 0.3f;
    [SerializeField] private float pulseDuration = 1.5f;
    [SerializeField] private Color pulseColor = Color.cyan;

    [Header("VertexPulse - 등장 패턴")]
    [SerializeField, Range(0f, 1f)] private float pulseSpawnChance = 0.3f; // 사이클마다 버텍스가 등장할 확률
    [SerializeField] private float pulseIntervalMin = 1.5f; // 다음 시도까지 최소 대기 시간
    [SerializeField] private float pulseIntervalMax = 3.5f;// 다음 시도까지 최대 대기 시간 

    private MeshFilter meshFilter;
    private MeshRenderer meshRenderer;
    private Mesh circleMesh;
    private Camera cam;

    private void Awake()
    {
        meshFilter = GetComponent<MeshFilter>();
        meshRenderer = GetComponent<MeshRenderer>();
    }

    public void Build()
    {
        var (verts, triangles) = TitleSolid.Icosphere(subdivisions, solidScale);
        var edges = TitleSolid.BuildEdgesFromTriangles(triangles);
        var mesh = TitleSolid.BuildLineMesh(verts, edges);
        mesh.name = "IcosahedronWireFrame";

        meshFilter.mesh = mesh;
        if (lineMaterial != null)
            meshRenderer.material = lineMaterial;

        SpawnVertexPulses(verts);
    }

    private void SpawnVertexPulses(List<Vector3> verts)
    {
        if (pulseMaterial == null)
        {
            Debug.LogWarning("[SolidBuilder] pulseMaterial이 없어 버텍스 펄스를 생성하지 않습니다.");
            return;
        }

        circleMesh = TitleSolid.BuildCircleMesh(circleSegments);
        cam = targetCamera != null ? targetCamera : Camera.main;

        // 펄스 타이밍을 어긋나게 하기 위해, 버텍스마다 독립된 루틴을 가짐
        foreach (var v in verts)
        {
            VertexPulse pulse = CreatePulseObject(v);
            StartCoroutine(VertexPulseRoutine(pulse));
        }
    }

    private IEnumerator VertexPulseRoutine(VertexPulse pulse)
    {
        // 버텍스마다 랜덤 초기 대기
        yield return new WaitForSeconds(Random.Range(0f, pulseIntervalMax));

        while (true)
        {
            if (Random.value <= pulseSpawnChance)
            {
                pulse.Play();
                yield return new WaitForSeconds(pulseDuration); // 생성된 경우는 펄스 시간만큼 대기
            }

            yield return new WaitForSeconds(Random.Range(pulseIntervalMin, pulseIntervalMax));
        }
    }

    private VertexPulse CreatePulseObject(Vector3 localPos)
    {
        GameObject pulseObj = new GameObject("VertexPulse");
        pulseObj.transform.SetParent(transform, false); // 펄스를 solid의 자식으로 둬서 회전을 따라가게 함
        pulseObj.transform.localPosition = localPos;

        VertexPulse pulse = pulseObj.AddComponent<VertexPulse>();
        pulse.Init(circleMesh, pulseMaterial, cam, pulseMaxRadius, pulseDuration, pulseColor);

        return pulse;
    }
}
```

#### SolidRotator.cs
```cs
using System.Collections;
using UnityEngine;

public class SolidRotator : MonoBehaviour
{
    [Header("Solid - 회전 설정")]
    [SerializeField] private float rotationSpeed = 20f;           // 기본 회전 속도(도/초)
    [SerializeField] private float rotationSpeedVariance = 8f;    // 속도에 랜덤 편차
    [SerializeField] private float directionChangeInterval = 4f;  // 한 방향 유지 시간
    [SerializeField] private float directionTransitionTime = 2f;  // 방향 전환에 걸리는 시간

    private Vector3 currentAxis;
    private float currentRotationSpeed;

    private void Start()
    {
        currentAxis = Random.onUnitSphere;
        currentRotationSpeed = rotationSpeed;
        StartCoroutine(RotationDirectionRoutine());
    }


    private void Update()
    {
        transform.Rotate(currentAxis, currentRotationSpeed * Time.deltaTime, Space.World);
    }

    private IEnumerator RotationDirectionRoutine()
    {
        while (true)
        {
            yield return new WaitForSeconds(directionChangeInterval);

            Vector3 newAxis = Random.onUnitSphere;
            float newSpeed = rotationSpeed + Random.Range(-rotationSpeedVariance, rotationSpeedVariance);

            Vector3 startAxis = currentAxis;
            float startSpeed = currentRotationSpeed;
            float t = 0f;

            while (t < directionTransitionTime)
            {
                t += Time.deltaTime;
                float k = t / directionTransitionTime;
                currentAxis = Vector3.Slerp(startAxis, newAxis, k).normalized;
                currentRotationSpeed = Mathf.Lerp(startSpeed, newSpeed, k);
                yield return null;
            }

            currentAxis = newAxis;
            currentRotationSpeed = newSpeed;
        }
    }
}
```

#### SceneLoader 수정
- `LoadScene()` 및 `LoadStage()`가 둘 다 `public`이라서 헷갈린다. 세부적인 구현도 다른 상황.

2. 실질적인 기능은 `private`으로 숨김

#### GameManagement 분리
- 정말로 게임 전체에서 사용하는 것들과, 각 세이브파일마다 달라져야 하는 것들(메인메뉴 씬에 진입했을 때 초기화돼야 하는 것들)을 구분하기로 함

>[!note]
>- `GameManagement` : 게임 전체에서 관리하는 것들
>- `GameSession` : 각 세이브 파일로 `MainMenuScene`에 들어간 이후에 관리되는 것들

- 판단 기준을 좀 명확히 잡아놓는 게 좋겠음.
	- 게임 전체의 규칙을 놓는 위치(성장, 보상 등등)
	- `TitleScene`에서 필요한지 여부

지금 **세션을 나눈 이유는 "세이브 파일"에 종속되는지 여부**였음. 그것만 생각하면 됨.

1. 전체적인 계산 로직이나 규칙은 `GameManagement`에 유지하는 게 맞음 
	- 세이브 파일에 관계 없이 항상 게임 전체에 적용되는 요소이기 때문.
2. 진행도 관련 로직은 `Session`에서 갖고 있는 게 맞다.

## 생각보다 대공사가 됐다

### 혼란하니까 뭐가 필요한지 정리

- 기존 
	- 별도의 세이브 슬롯 관리 X
	- 시작점이 `MainMenuScene`이었고, `StageScene`과의 씬 전환만 관리하면 됐음

- 현재
	- 별도의 세이브 슬롯 관리 O(아직 구현 X)
	- 시작점이 `TitleScene`, `시작하기`나 `계속하기` 클릭 시 세션 단위에서 진행 데이터를 초기화 / 불러오기할 필요가 생김
		- 개별 세션 단위의 데이터 관리는 이미 되어 있다고 봐야 함. 
		- 타이틀에서 메인메뉴로 넘어갈 때 데이터를 넘겨주는 로직이 필요해보인다. 이를 위해 `GameManagement`의 데이터 관리 매니저와 `GameSession`의 데이터 관리 매니저를 분리할 필요가 생겼음.

### 데이터 관리 매니저 분리
- 일단 `PlayerDataManager`의 구조를 둘로 나눴다.
```
GameManagement
- GameDataManager (전체적인 게임에서 사용하는 데이터)
  
GameSession
- PlayerSessionDataManager (진행 정보 저장)
```

분리 자체는 AI의 도움을 받음. 모든 스크립트를 넘겨주지 않고 있기에 중간중간 수정하고 있다.
